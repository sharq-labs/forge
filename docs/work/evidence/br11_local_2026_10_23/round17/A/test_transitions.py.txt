import itertools
from unittest.mock import patch
import pytest
from engcore.mind.world import BudgetReader, OwnerKind
from engcore.mind.world.authority import OwnerUnavailable
from engcore.sria.campaign.budget import BudgetLedger
from tests.mind.test_authority_review_r14 import _settle
from tests.mind.test_authority_review_r16 import _interrupt_handing

@pytest.mark.parametrize("order", list(itertools.permutations(range(3))))
def test_three_detached_prefixes_any_order(order):
    ledger=BudgetLedger(total_budget=100.)
    _settle(ledger,"initial")
    snapshots=[BudgetLedger.from_dict(ledger.to_dict())]
    for label in ("one","two"):
        _settle(ledger,label)
        snapshots.append(BudgetLedger.from_dict(ledger.to_dict()))
    held=[snapshots[0]]
    reader=BudgetReader.following(lambda: ({"a":BudgetLedger.from_dict(held[0].to_dict())},{}))
    for i in order:
        held[0]=snapshots[i]
        _interrupt_handing(reader)
    held[0]=snapshots[-1]
    assert reader.read(OwnerKind.BUDGET_LEDGER,"a") is not None
    assert not reader._diverged
    held[0]=snapshots[0]
    with pytest.raises(OwnerUnavailable): reader.read(OwnerKind.BUDGET_LEDGER,"a")

@pytest.mark.parametrize("order", list(itertools.permutations(range(3))))
def test_three_detached_branches_any_order(order):
    ledger=BudgetLedger(total_budget=100.)
    _settle(ledger,"initial")
    snapshots=[BudgetLedger.from_dict(ledger.to_dict()) for _ in range(3)]
    for i,s in enumerate(snapshots): _settle(s,str(i))
    held=[ledger]
    reader=BudgetReader.following(lambda: ({"a":BudgetLedger.from_dict(held[0].to_dict())},{}))
    for i in order:
        held[0]=snapshots[i]
        _interrupt_handing(reader)
    for s in snapshots:
        held[0]=s
        with pytest.raises(OwnerUnavailable): reader.read(OwnerKind.BUDGET_LEDGER,"a")

def test_absent_account_still_reconciles_spend():
    ledger=BudgetLedger(total_budget=100.)
    _settle(ledger,"initial")
    held={"a":ledger}
    reader=BudgetReader.following(lambda: (held,{}))
    old=BudgetLedger.from_dict(ledger.to_dict())
    _settle(ledger,"one")
    _interrupt_handing(reader)
    held.clear()
    assert reader.read(OwnerKind.BUDGET_LEDGER,"a") is None
    held["a"]=old
    with pytest.raises(OwnerUnavailable): reader.read(OwnerKind.BUDGET_LEDGER,"a")
