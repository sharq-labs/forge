import json
from unittest.mock import patch
import pytest
from engcore.mind.world import BudgetReader, OwnerKind
from engcore.mind.world.authority import OwnerUnavailable
from engcore.sria.campaign.budget import BudgetLedger, BudgetCharge
from engcore.scientific.experiments.experiment import ScientificExperiment
from engcore.scientific.experiments.evaluation import ScientificEvaluation, EvaluationStatus
from engcore.scientific import to_json
from tests.mind.test_authority_review_r14 import _desk, _solver, _settle, _refused, ax
from tests.mind.test_authority_review_r16 import _interrupt_handing

@pytest.mark.parametrize("alias",["A1","B1","C1","D1","E1","F1"])
@pytest.mark.parametrize("failures,restore",[(2,True),(1,True),(2,False)])
def test_capture(alias,failures,restore):
    o=ax.owned("battery"); lid=_solver(o); calls=[]
    def execute(auth):
        calls.append(1); _settle(o.ledgers[lid],f"run.{len(calls)}")
    reader,_,gateway,admitted=_desk(o,lambda:(o.ledgers,o.experiments),execute)
    checkpoint=BudgetLedger.from_dict(o.ledgers[lid].to_dict())
    gateway.dispatch(admitted.admitted)
    original=BudgetReader._retain_handing; hits=[]
    def cut(mapping):
        hits.append(1)
        if len(hits)<=failures: raise KeyboardInterrupt()
        return original(mapping)
    with patch.object(BudgetReader,"_retain_handing",staticmethod(cut)):
        with pytest.raises(KeyboardInterrupt):reader.pinned()
    if failures==2:
        pending=list(reader._uncompared.values())
        assert len(pending)==1 and type(pending[0]) is tuple
        assert pending[0][0] is o.ledgers
    if restore:o.ledgers[lid]=checkpoint
    # Honest unchanged recovery must remain readable, without permanent retirement.
    if not restore:
        assert reader.read(OwnerKind.BUDGET_LEDGER,lid) is not None
        assert not reader._diverged and reader._sealed is None
    refused=_refused(gateway,admitted.admitted)
    bug=failures==2 and restore
    print(json.dumps(dict(alias=alias,failures=failures,restore=restore,refused=refused,calls=len(calls),sealed=reader._sealed)))
    assert refused is (not bug)
    assert len(calls)==(2 if bug else 1)

@pytest.mark.parametrize("kind",["ledger","experiment"])
@pytest.mark.parametrize("keys",[{2:"a",10:"b"},{"2":"a","10":"b"}])
@pytest.mark.parametrize("json_correction",[False,True])
def test_normalization(kind,keys,json_correction):
    if kind=="ledger":
        account=BudgetLedger(total_budget=100.); name="a"; owner_kind=OwnerKind.BUDGET_LEDGER; entry_owner=BudgetCharge
        held=[account]; follow=lambda:({name:held[0]}, {})
        reader=BudgetReader.following(follow)
        account.settle(charge_id="new",action_id="new",iteration=1,family="explore",realized=1.,detail={"nested":keys})
        clone=lambda:BudgetLedger.from_dict(json.loads(json.dumps(account.to_dict(),sort_keys=True,default=str)))
    else:
        o=ax.owned("battery"); name,account=next(iter(o.experiments.items()));owner_kind=OwnerKind.EXPERIMENT;entry_owner=ScientificEvaluation
        held=[account]; follow=lambda:({}, {name:held[0]})
        reader=BudgetReader.following(follow)
        account.record(ScientificEvaluation("new",{},status=EvaluationStatus.NOT_CONVERGED,metadata={"nested":keys}))
        clone=lambda:ScientificExperiment.from_dict(json.loads(to_json(account)))
    _interrupt_handing(reader);held[0]=clone()
    fresh=BudgetReader.following(follow); assert fresh.read(owner_kind,name) is not None
    real=BudgetReader._history
    def corrected(items):
        return real(entry_owner.from_dict(json.loads(json.dumps(x.to_dict(),sort_keys=True,default=str))) for x in items)
    with patch.object(BudgetReader,"_history",staticmethod(corrected if json_correction else real)):
        bug=not json_correction and 2 in keys
        if bug:
            for _ in range(2):
                with pytest.raises(OwnerUnavailable,match="rewrote its history"):reader.read(owner_kind,name)
            assert reader._diverged
        else:
            assert reader.read(owner_kind,name) is not None
            assert not reader._diverged
    print(json.dumps(dict(kind=kind,numeric=2 in keys,json_correction=json_correction,false_retired=bool(reader._diverged))))
