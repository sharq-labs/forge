import threading
from unittest.mock import patch
import pytest
from engcore.mind.world import BudgetReader, OwnerKind
from engcore.mind.world.authority import OwnerUnavailable
from engcore.sria.campaign.budget import BudgetLedger
from tests.mind.test_authority_review_r14 import _settle
from tests.mind.test_authority_review_r16 import _interrupt_handing

def reader_pending(n=2):
    account=BudgetLedger(total_budget=100.)
    _settle(account, "initial")
    held=[({"a": account}, {})]
    def follow():
        if held[0] is None: return None
        return ({"a": BudgetLedger.from_dict(account.to_dict())}, {})
    reader=BudgetReader.following(follow)
    _settle(account,"new")
    for _ in range(n): _interrupt_handing(reader)
    return reader, held

def test_seal_during_unlocked_reconciliation_refuses_worker():
    reader, held=reader_pending()
    entered,resume=threading.Event(),threading.Event()
    local=threading.local()
    real_read,real_history=BudgetReader._read_back,BudgetReader._history
    outcome=[]
    def read(s):
        result=real_read(s)
        if threading.current_thread() is worker: local.reconcile=True
        return result
    def history(items):
        if getattr(local,"reconcile",False):
            local.reconcile=False;entered.set();assert resume.wait(5)
        return real_history(items)
    def run():
        try: outcome.append(reader.read(OwnerKind.BUDGET_LEDGER,"a"))
        except BaseException as e: outcome.append(e)
    worker=threading.Thread(target=run)
    with patch.object(BudgetReader,"_read_back",read),patch.object(BudgetReader,"_history",staticmethod(history)):
        worker.start()
        try:
            assert entered.wait(5)
            held[0]=None
            with pytest.raises(OwnerUnavailable): reader.pinned()
            assert reader._sealed is not None
        finally: resume.set();worker.join(5)
    assert not worker.is_alive()
    assert len(outcome)==1 and isinstance(outcome[0],OwnerUnavailable)

def test_reconcile_interrupt_preserves_barrier_then_recovers():
    reader,_=reader_pending()
    real=BudgetReader._read_back
    patches=[]
    def read(s):
        result=real(s)
        p=patch.object(BudgetReader,"_history",side_effect=KeyboardInterrupt)
        p.start();patches.append(p)
        return result
    try:
        with patch.object(BudgetReader,"_read_back",read):
            with pytest.raises(OwnerUnavailable): reader.read(OwnerKind.BUDGET_LEDGER,"a")
    finally:
        for p in patches:p.stop()
    assert reader._unread_entries and not reader._diverged and reader._sealed is None
    assert reader.read(OwnerKind.BUDGET_LEDGER,"a") is not None

def test_many_detached_interruptions_clear_after_reconciliation():
    reader,_=reader_pending(40)
    assert len(reader._unread_entries[("ledger","a")].observations)==40
    assert reader.read(OwnerKind.BUDGET_LEDGER,"a") is not None
    assert not reader._unread_entries and not reader._uncompared and not reader._in_flight
